local function safe(name,fn,...)
    local args=table.pack(...)
    local result=table.pack(pcall(function() return fn(table.unpack(args,1,args.n)) end))
    if not result[1] then warn("[Vy SpaceMine] "..name..": "..tostring(result[2])) end
    return table.unpack(result,1,result.n)
end
local function http(options)
    local httpreq=request or http_request or syn and syn.request
    assert(type(httpreq)=="function","Executor request() is unavailable")
    local response=httpreq(options)
    assert(type(response)=="table","HTTP request returned invalid response")
    assert(tonumber(response.StatusCode),"HTTP response had no status code")
    return response
end

local function sendWebhook(reason,force)
    if Config.WebhookURL=="" or State.WebhookBusy then return end
    if not Config.WebhookURL:match("^https://discord%.com/api/webhooks/%d+/[%w_%-]+") then
        State.WebhookStatus="Invalid URL"
        return
    end
    if os.clock()-State.LastWebhook<(force and 3 or Config.WebhookInterval) then return end
    local gems=State.Gems.Onyx+State.Gems.Sunstone
    if not force and gems==State.LastWebhookGems then return end
    State.WebhookBusy=true
    State.LastWebhook=os.clock()
    local ok,response=pcall(http,{Url=Config.WebhookURL..(Config.WebhookURL:find("?",1,true) and "&wait=true" or "?wait=true"),Method="POST",Headers={["Content-Type"]="application/json"},Body=HttpService:JSONEncode(webhookPayload(reason))})
    if ok and response.StatusCode>=200 and response.StatusCode<300 then
        State.WebhookStatus="Sent "..os.date("!%H:%M:%S").." UTC"
        State.LastWebhookGems=gems
    elseif ok then
        State.WebhookStatus="HTTP "..tostring(response.StatusCode)
        if response.StatusCode==429 then
            local decodedOK,data=pcall(function() return HttpService:JSONDecode(response.Body) end)
            if decodedOK then State.LastWebhook=os.clock()+math.min(300,tonumber(data.retry_after) or 5) end
        end
    else
        State.WebhookStatus="Failed"
        warn("[Vy SpaceMine] Webhook: "..tostring(response))
    end
    State.WebhookBusy=false
end
local function releaseRoot()
    if State.Hold then State.Hold:Destroy() State.Hold=nil end
    if State.HoldAttachment then State.HoldAttachment:Destroy() State.HoldAttachment=nil end
    local root=State.AnchoredRoot
    State.AnchoredRoot=nil
    if root and root.Parent then root.Anchored=State.WasAnchored end
    for part,canCollide in pairs(State.CollisionParts) do
        if part.Parent then part.CanCollide=canCollide end
    end
    table.clear(State.CollisionParts)
    if State.HeldHumanoid and State.HeldHumanoid.Parent then State.HeldHumanoid.AutoRotate=State.WasAutoRotate end
    State.HeldHumanoid=nil
    State.WasAutoRotate=nil
    State.HeldTarget=nil
end

local function miningActive()

local function readInventory()
    local save=Modules.Save.Get()
    if not save or not save.Inventory or not save.Inventory.Misc then return nil end
    local totals={Onyx=0,Sunstone=0}
    for _,item in pairs(save.Inventory.Misc) do
        local kind=GEM_IDS[item.id]
        if kind then totals[kind]=totals[kind]+(tonumber(item._am) or 1) end
    end
    return totals
end
local function updateInventory()
    local totals=readInventory()
    if not totals then return end
    if State.LastInventory then
        for kind,amount in pairs(totals) do
            State.Gems[kind]=State.Gems[kind]+math.max(0,amount-(State.LastInventory[kind] or 0))
        end
    end
    State.LastInventory=totals
end
local function scan()
    local world=Modules.World.GetLocal()
    State.World=world
    State.Counts={Onyx=0,Sunstone=0}
    State.OnlyStuckLeft=false
    if Modules.Instance.GetInstanceID()~=INSTANCE_ID or not world or world.Destroyed or not world.Model or not world.Model.Parent or world.Respawning then
        clearTarget()
        return false
    end
    local root=rootPart()
    if not root then clearTarget() return false end
    local now=os.clock()
    local current=State.Target
    local currentKind=current and validTarget(world,current) and TARGET_IDS[current:GetDirectory()._id] or nil
    local stuckAfter=currentKind and Config.StuckTimeout[currentKind] or nil
    if stuckAfter and not current.TempBroken and now-State.TargetSince>=stuckAfter then
        State.Skipped[current]=now+Config.SkipDuration
        State.SkipTotal=State.SkipTotal+1
        State.Target=nil
        State.TargetKind=nil
    end
    local nearest=nil
    local nearestDistance=nil
    local bestPriority=nil
    local numberOfBlocks=0
    local skippedLeft=0
    for _,block in pairs(world.Blocks) do
        numberOfBlocks=numberOfBlocks+1
        local directory=block:GetDirectory()
        local kind=directory and TARGET_IDS[directory._id] or nil
        if kind and block:GetPart().Parent then
            State.Counts[kind]=State.Counts[kind]+1
            if not block.TempBroken then
                local skipUntil=State.Skipped[block]
                if skipUntil and skipUntil>now then
                    skippedLeft=skippedLeft+1
                else
                    local distance=(root.Position-block:GetCFrame().Position).Magnitude
                    local priority=kind=="Onyx" and 1 or 2
                    if not bestPriority or priority<bestPriority or priority==bestPriority and distance<nearestDistance then
                        nearest=block
                        nearestDistance=distance
                        bestPriority=priority
                    end
                end
            end
        end
    end
    local switchToOnyx=nearest and bestPriority==1 and State.Target and State.Target:GetDirectory()._id~="Onyx"
    if not validTarget(world,State.Target) or switchToOnyx then State.Target=nearest end
    State.TargetKind=State.Target and TARGET_IDS[State.Target:GetDirectory()._id] or nil
    if State.Target~=State.TimedTarget then
        State.TimedTarget=State.Target
        State.TargetSince=now
    end
    State.OnlyStuckLeft=nearest==nil and skippedLeft>0
    return numberOfBlocks>0
end
local function holdTarget()
    if not miningActive() then releaseRoot() return end
    local root=rootPart()
    if not root then releaseRoot() return end
    if State.AnchoredRoot~=root then
        releaseRoot()
        State.AnchoredRoot=root
        State.WasAnchored=root.Anchored
        local attachment=Instance.new("Attachment")
        attachment.Name="MineHoldAttachment"
        attachment.Parent=root
        local hold=Instance.new("AlignPosition")
        hold.Name="MineHold"
        hold.Mode=Enum.PositionAlignmentMode.OneAttachment
        hold.Attachment0=attachment
        hold.MaxForce=math.max(10000,root.AssemblyMass*Workspace.Gravity*15)
        hold.MaxVelocity=20
        hold.Responsiveness=35
        hold.RigidityEnabled=false
        hold.ApplyAtCenterOfMass=true
        hold.Parent=root
        State.HoldAttachment=attachment
        State.Hold=hold
    end
    for _,part in ipairs(root.Parent:GetDescendants()) do
        if part:IsA("BasePart") then
            if State.CollisionParts[part]==nil then State.CollisionParts[part]=part.CanCollide end
            part.CanCollide=false
        end
    end
    local humanoid=root.Parent:FindFirstChildOfClass("Humanoid")
    if humanoid and State.HeldHumanoid~=humanoid then
        State.HeldHumanoid=humanoid
        State.WasAutoRotate=humanoid.AutoRotate
    end
    if humanoid then humanoid.AutoRotate=false end
    root.Anchored=false
    local targetFrame=State.Target:GetCFrame()*CFrame.new(0,6,0)
    State.Hold.Position=targetFrame.Position
    local now=os.clock()
    local distance=(root.Position-targetFrame.Position).Magnitude
    if State.HeldTarget~=State.Target or distance>12 and now-State.LastRecovery>=1 then
        root.CFrame=targetFrame
        root.AssemblyLinearVelocity=Vector3.zero
        root.AssemblyAngularVelocity=Vector3.zero
        State.LastRecovery=now
    end
    if root.AssemblyAngularVelocity.Magnitude>2 then root.AssemblyAngularVelocity=Vector3.zero end
    State.HeldTarget=State.Target
end

local function attemptPendingTeleport()
    local pending=State.PendingTeleport
    if not pending or State.Teleporting or os.clock()<State.TeleportRetryAt then return false end
    State.Teleporting=true
    State.IsHopping=true
    State.TeleportStartedAt=os.clock()
    State.Status="Teleporting..."
    refreshUI()
    local ok=pcall(TeleportService.TeleportToPlaceInstance,TeleportService,pending.PlaceID,pending.JobID,LocalPlayer)
    if not ok then
        State.Teleporting=false
        State.TeleportRetryAt=os.clock()+3
        State.TeleportFailures=State.TeleportFailures+1
        if State.TeleportFailures>=3 then
            State.PendingTeleport=nil
            State.TeleportFailures=0
            State.IsHopping=false
            State.Status="Teleport failed - finding another server"
        else
            State.Status="Teleport retry in 3s"
        end
        refreshUI()
        return false
    end
    return true
end
local function HopServer()
    if State.IsHopping or State.PendingTeleport or State.Teleporting then return end
    State.IsHopping=true
    clearTarget()
    State.Status="Finding server..."
    refreshUI()
    task.spawn(function()
        local httpreq=request or http_request or syn and syn.request
        local ok,data=pcall(function()
            local url=string.format("https://games.roblox.com/v1/games/%s/servers/0?sortOrder=Asc&excludeFullGames=true&limit=100",tostring(PlaceId))
            if httpreq then return HttpService:JSONDecode(httpreq({Url=url,Method="GET"}).Body) end
            return HttpService:JSONDecode(game:HttpGet(url))
        end)
        if ok and data and data.data then
            local preferred={}
            local fallback={}
            for _,server in ipairs(data.data) do
                if type(server)=="table" and server.id~=JobId and server.playing and server.maxPlayers and server.playing<server.maxPlayers then
                    table.insert(fallback,server.id)
                    if server.playing>=Config.MinPlayers and server.playing<=Config.MaxPlayers then table.insert(preferred,server.id) end
                end
            end
            local candidates=#preferred>0 and preferred or fallback
            if #candidates>0 then
                State.TotalHops=State.TotalHops+1
                safe("Inventory before hop",updateInventory)
                task.spawn(function() safe("Webhook before hop",sendWebhook,"Server hop",true) end)
                State.PendingTeleport={PlaceID=PlaceId,JobID=tostring(candidates[math.random(1,#candidates)])}
                State.TeleportRetryAt=0
                State.TeleportFailures=0
                State.Status="Teleport queued..."
                refreshUI()
                attemptPendingTeleport()
                return
            end
        end
        State.Status="No different server found"
        State.IsHopping=false
        refreshUI()
    end)
end

local function enterSpaceMine()
    if Modules.Instance.GetInstanceID()==INSTANCE_ID then
        State.InSpaceMine=true
        return
    end
    State.Status="Entering Space Mine..."
    refreshUI()
    assert(type(setthreadidentity)=="function","setthreadidentity unavailable")
    local finished=false
    local enterOK=false
    local enterErr=nil
    task.spawn(function()
        local identityOK,identityErr=pcall(setthreadidentity,2)
        if identityOK then
            enterOK,enterErr=pcall(function() I.Enter("SpaceMiningEvent") end)
            pcall(setthreadidentity,7)
        else
            enterErr=identityErr
        end
        finished=true
    end)
    local deadline=os.clock()+15
    while not finished and os.clock()<deadline do task.wait(0.05) end
    assert(finished,"Space Mine Enter call timed out")
    assert(enterOK,enterErr)
end
local function waitForSpaceMineWorld()

        local world=Modules.World.GetLocal()

            State.InSpaceMine=true
            return true
        end
        State.Status="Waiting for Space Mine world..."
        refreshUI()
        task.wait(0.25)
    end
    return false
end

    Modules.Selection=require(client.ToolCmds.BlockInteraction)
    Modules.Auto.Disable()
    local inventoryDeadline=os.clock()+30
    while not Modules.Save.Get() and os.clock()<inventoryDeadline do task.wait(0.25) end
    updateInventory()
    enterSpaceMine()
    assert(waitForSpaceMineWorld(),"Space Mine world did not load")
    assert(teleportZone8AndWaitBlocks(),"Zone 8 blocks did not load")
    installMiningHooks()
    State.Ready=true
    State.Status="Scanning Onyx + Sunstone"
    refreshUI()
end
local ok,err=pcall(startup)
if not ok then
    State.Status="Startup error"
    warn("[Vy SpaceMine] "..tostring(err))
    refreshUI()
    return
end
local function stop()
    if not State.Alive then return end
    State.Alive=false
    clearTarget()
    if InputHook and type(restorefunction)=="function" then pcall(restorefunction,InputHook) InputHook=nil end
    if SelectionHook and type(restorefunction)=="function" then pcall(restorefunction,SelectionHook) SelectionHook=nil end
    for _,connection in ipairs(Connections) do pcall(function() connection:Disconnect() end) end
    table.clear(Connections)
    if ScreenGui and ScreenGui.Parent then ScreenGui:Destroy() end
end
getgenv().VyOnyxController={Stop=stop,State=State,Config=Config}
task.spawn(function()
    while State.Alive do
        if State.Teleporting then
            if os.clock()-State.TeleportStartedAt>=20 then
                State.Teleporting=false
                State.TeleportRetryAt=os.clock()+3
                State.Status="Teleport timeout - retrying"
                refreshUI()
            end
        elseif State.PendingTeleport then
            attemptPendingTeleport()
        end
        task.wait(0.5)
    end
end)
task.spawn(function()
    while State.Alive do
        task.wait(0.25)
        safe("Inventory",updateInventory)
        refreshUI()
    end
end)
task.spawn(function()
    while State.Alive do
        task.wait(1)
        if State.Ready then safe("Webhook",sendWebhook,"Mining update",false) end
    end
end)
task.spawn(function()
    while State.Alive do
        task.wait(Config.ScanInterval)
        if State.IsHopping then continue end
        if not Config.MineEnabled then clearTarget() continue end
        Modules.Auto.Disable()
        local loaded=scan()
        if not loaded then
            State.Status="Waiting for Space Mine blocks..."
            continue
        end
        if State.Counts.Onyx+State.Counts.Sunstone==0 then
            clearTarget()
            State.Status=Config.HopEnabled and "No Onyx/Sunstone - hopping" or "No Onyx/Sunstone"
            refreshUI()
            if Config.HopEnabled then HopServer() end
            continue
        end
        if State.OnlyStuckLeft then
            clearTarget()
            if Config.HopEnabled then
                State.Status="Only stuck ore left - hopping"
                refreshUI()
                HopServer()
            else
                table.clear(State.Skipped)
                State.Status="Only stuck ore left - retrying"
                refreshUI()
            end
            continue
        end
        if State.Target then
            State.Status="Mining "..tostring(State.TargetKind)..(State.SkipTotal>0 and " | skipped "..State.SkipTotal or "")
            holdTarget()
            if not State.Target.TempBroken then
                State.World:LocalSetTarget(State.Target)
                Modules.Input.Pressed:FireAsync()
            end
        else
            State.Status="Waiting for ore break confirmation..."
        end
        refreshUI()
    end
end)
